## Code Example: Real-Time Handler

```typescript
// src/websocket/ws-handler.ts (auto-generated)
export class WebSocketHandler {
  constructor(private subscriptionService: SubscriptionService) {}
  
  async handleSubscribe(socket: Socket, data: { topic: string }) {
    const subscription = await this.subscriptionService.subscribe(
      socket.id,
      data.topic
    );
    
    // Add socket to room
    socket.join(`topic:${data.topic}`);
    
    socket.emit('subscribed', {
      topic: data.topic,
      subscriptionId: subscription.id,
    });
  }
  
  async notifyUpdate(topic: string, update: any) {
    // Broadcast to all subscribers
    this.io.to(`topic:${topic}`).emit('update', update);
  }
}
```
